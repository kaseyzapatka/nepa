def page_num_int(value):
    """Extract first numeric token from page labels like '1', 'Page-1', or '1-6'."""
    if pd.isna(value):
        return pd.NA
    match = re.search(r"(\d+)", str(value))
    if match:
        return int(match.group(1))
    return pd.NA

def list_documents(project_id):
    """List all documents for a given project."""
    docs = ce_documents[ce_documents["project_id"] == project_id]
    if len(docs) == 0:
        print(f"No documents found for project_id: {project_id}")
        return None
    
    print(f"\nDocuments for project: {project_id}")
    print("-" * 80)
    display_cols = ["document_id", "document_type", "document_type_category", 
                    "main_document", "file_name", "total_pages"]
    available_cols = [c for c in display_cols if c in docs.columns]
    return docs[available_cols]
def view_page(document_id, page_num=1):
    """View a specific page from a document."""
    page_target = page_num_int(page_num)
    if pd.isna(page_target):
        print(f"Invalid page number: {page_num}")
        return None

    page = CE_pages[(CE_pages["document_id"] == document_id) & 
                    (CE_pages["_page_number_num"] == page_target)]
    
    if len(page) == 0:
        print(f"Page {page_num} not found for document_id: {document_id}")
        return None
    
    # Get document metadata
    doc = ce_documents[ce_documents["document_id"] == document_id]
    if len(doc) > 0:
        print(f"\n{'='*80}")
        print(f"Document: {doc.iloc[0].get('file_name', 'N/A')}")
        print(f"Type: {doc.iloc[0].get('document_type', 'N/A')} | "
              f"Category: {doc.iloc[0].get('document_type_category', 'N/A')} | "
              f"Main: {doc.iloc[0].get('main_document', 'N/A')}")
        print(f"{'='*80}")
    
    print(f"\n--- PAGE {page_num} ---\n")
    print(page.iloc[0]["page_text"])
    print(f"\n--- END PAGE {page_num} ---\n")
    return page.iloc[0]["page_text"]
def view_pages(document_id, start_page=1, end_page=5):
    """View a range of pages from a document."""
    # Get document metadata
    doc = ce_documents[ce_documents["document_id"] == document_id]
    if len(doc) > 0:
        print(f"\n{'='*80}")
        print(f"Document: {doc.iloc[0].get('file_name', 'N/A')}")
        print(f"Type: {doc.iloc[0].get('document_type', 'N/A')} | "
              f"Category: {doc.iloc[0].get('document_type_category', 'N/A')} | "
              f"Main: {doc.iloc[0].get('main_document', 'N/A')}")
        print(f"{'='*80}")
    
    for page_num in range(start_page, end_page + 1):
        page = CE_pages[(CE_pages["document_id"] == document_id) & 
                        (CE_pages["_page_number_num"] == page_num)]
        if len(page) > 0:
            print(f"\n--- PAGE {page_num} ---\n")
            print(page.iloc[0]["page_text"])
            print(f"\n--- END PAGE {page_num} ---\n")
        else:
            print(f"Page {page_num} not found")
            break
def view_project_pages(project_id, document_id=None):
    """View all pages of EISch document in a project (no truncation).
    
    Args:
        project_id: The project ID to filter documents
        document_id: Optional document ID to view only that specific document
    """
    docs = ce_documents[ce_documents["project_id"] == project_id]
    
    # Filter by document_id if provided
    if document_id is not None:
        docs = docs[docs["document_id"] == document_id]
        
    if len(docs) == 0:
        if document_id is not None:
            print(f"No document found with document_id: {document_id} in project_id: {project_id}")
        else:
            print(f"No documents found for project_id: {project_id}")
        return
        
    for _, doc in docs.iterrows():
        doc_id = doc["document_id"]
        print(f"\n{'='*80}")
        print(f"Document: {doc.get('file_name', 'N/A')}")
        print(
            f"Type: {doc.get('document_type', 'N/A')} | "
            f"Category: {doc.get('document_type_category', 'N/A')} | "
            f"Main: {doc.get('main_document', 'N/A')}"

        print(f"{'='*80}")
                
        doc_pages = (
            CE_pages[CE_pages["document_id"] == doc_id]
            .sort_values(["_page_number_num", "page_number"], na_position="last")
        )
                
        for _, page in doc_pages.iterrows():
            print(f"\n--- PAGE {page['page_number']} ---\n")
            print(page["page_text"])
def search_pages(project_id, search_term):
    """Search for a term in all pages of a project."""
    docs = ce_documents[ce_documents["project_id"] == project_id]
    doc_ids = docs["document_id"].tolist()
    
    matching_pages = ce_pages[
        (ce_pages["document_id"].isin(doc_ids)) & 
        (ce_pages["page_text"].str.contains(search_term, case=False, na=False))
    ]
    
    print(f"\nFound {len(matching_pages)} pages containing '{search_term}'")
    print("-" * 80)
    
    for _, page in matching_pages.iterrows():
        doc = docs[docs["document_id"] == page["document_id"]].iloc[0]
        print(f"\nDocument: {doc.get('file_name', 'N/A')} | Page: {page['page_number']}")
        
        # Show context around the search term
        text = page["page_text"]
        idx = text.lower().find(search_term.lower())
        if idx >= 0:
            start = max(0, idx - 200)
            end = min(len(text), idx + len(search_term) + 200)
            context = text[start:end]
            if start > 0:
                context = "..." + context
            if end < len(text):
                context = context + "..."
            print(context)
    
    return matching_pages
def get_random_project(n_docs=2):
    """Get a random project with at least n_docs documents."""
    import random
    doc_counts = ce_documents.groupby("project_id").size()
    eligible = doc_counts[doc_counts >= n_docs].index.tolist()
    
    project_id = random.choice(eligible)
    print(f"Random project with {doc_counts[project_id]} documents: {project_id}")
    return project_id

1 INTRODUCTION

The Southline Transmission Project (hereafter, Project), sponsored by Southline Transmission, L.L.C.
(Southline), a wholly owned subsidiary of Hunt Power, L.P., is a non-utility, merchant transmission line
designed to collect and transmit electricity across southern New Mexico and southern Arizona.

The Project consists of two sections: the New Build and the Upgrade sections. The New Build Section
is approximately 240 miles of new double-circuit 345-kilovolt (kV) transmission line along with
substation expansions, access roads, and associated infrastructure. The Upgrade Section is approximately
120 miles of Western Area Power Administration's (Western's or WAPA's) existing 115-kV
transmission line that is being upgraded to a double-circuit 230-kV transmission line as well as substation
expansions, access roads, and associated infrastructure.

This Geotechnical Investigation Plan of Development (Geotechnical POD) is for the Project's
Arizona New Build portion only (Figure 1). The total length of the Arizona alignment applicable to this
Geotechnical POD is approximately 72 miles, traversing Cochise County. Additionally, full Project
details can be found in the below referenced 2015 Project Final Environmental Impact Statement.

1.1 National Environmental Policy Act

On December 4, 2009, Southline submitted an “Application for Transportation and Utility Systems
and Facilities on Federal Lands” (Standard Form [SF] 299) to the Bureau of Land Management
(BLM) for a right-of-way (ROW), and then submitted an amended application on December 22, 2010.
Southline will use identified BLM-administered public land for a portion of the Project, in addition to
other federal, state, and private lands through New Mexico and Arizona. As part of the Project, Southline
will upgrade two existing lines in Arizona that are owned by WAPA. The BLM and WAPA are joint lead
agencies under the National Environmental Policy Act of 1969 (NEPA) regulations at 40 Code of Federal
Regulations (CFR) 1501.5(b).

BLM and WAPA jointly prepared the 2015 “Southline Transmission Line Project Final Environmental
Impact Statement” (Final EIS). A notice of intent (NOI) to prepare the EIS for the Project was published
in the Federal Register on April 4, 2012. The notice of availability (NOA) for the Draft EIS was published
on April 11, 2014 in the Federal Register. The NOA for the Final EIS was published on November 6,
2015 in the Federal Register. The BLM Record of Decision (ROD) was signed on April 21, 2016, and
WAPA'S ROD was signed April 5, 2016. The BLM and WAPA's Selected Alternative was the Agency
Preferred Alternative as described in the Final EIS. The BLM ROD authorized a ROW grant to Southline
to use public land administered by the BLM for the Project for 30 years.

outline the geotechnical investigation program, including information regarding the drilling, location of
boreholes, borehole drilling equipment, and borehole access.

Geotechnical investigations provide necessary information for the detailed engineering and design
of the transmission line, access roads, substations, and other facilities and are a required step in the
Project construction process. Southline, in coordination with the BLM, has developed an implementation
strategy that includes the preparation of a Geotechnical POD.

BORING WORK AREAS

A typical boring work area is 30 × 60 feet. This is enough room to get the drilling rig and support vehicles
in and work around the rig. There may be some minor disturbance outside that area in rare cases;
however, the maximum temporary disturbance at each boring site is anticipated to be less than 0.4 acres.
Surface disturbance within the boring work area may occur as a result of moving the drilling vehicles,
support vehicles, and associated equipment. Additional foot traffic will also occur around the drill
vehicles as the drill crew moves between the drill and support vehicles.

Work sites will be managed to minimize both environmental and physical impacts to the immediate work
area and surrounding area. All practical controls will be followed to minimize soil disturbance and
prevent soils from migrating offsite. Equipment and vehicles will be positioned to minimize damage to
vegetation and soils. Excavated soils and other work materials will be properly contained to minimize the
spread of material to outside the work area. If adverse weather conditions are anticipated, excavated soils
will be properly secured to prevent sediment migration offsite.

ACCESS ROUTES

Access roads to each boring will be restricted to existing paved or dirt roads with a minor amount
of overland travel. As described below in Section 2.2.2, no grading for access to geotechnical

2.1.3 Drilling Equipment Staging

All drilling equipment will be staged within the approved Project ROW. Equipment staging areas will
limit earth disturbance to the extent reasonable. Heavy-duty equipment will not be placed/stored over
exposed tree roots or within the critical root zone. At the end of each day the drilling vehicles and all
equipment will be left at the last completed boring location or moved to the next boring location.
Personnel will travel to and from the boring sites via four-wheel drive vehicles.

Existing roads will be used to access the majority of boring sites to facilitate the geotechnical

and aerial obstacles, marked underground utilities, and overhead utilities.

2.3 Identification of Potential Geologic Hazards

2.3.1 Overview of Geologic Hazards

A geological condition that has the potential to cause structural damage is classified as a geologic hazard
(or geohazard), and the presence of geohazards can pose significant risks to the Project. Geohazards
involve short-term processes, long-term processes, or both, and are influenced by both geological and
environmental conditions. Geohazards can be exacerbated by human activities. For these reasons, the
potential impacts of geohazards were analyzed as part of the Project NEPA process.

The NEPA analysis consisted of a preliminary geotechnical desktop study which identified potential
geohazards. Potential geohazards identified for the Project area are presented in Section 3.4.4 of the Final
EIS and include land subsidence, earth fissures, geological faults, earthquakes, landslides, volcanoes, and
karst. Analysis of these potential hazards is presented in Final EIS Section 4.4.4.

2.4 Geotechnical Exploration Methods

2.4.1 Geotechnical Drilling and Sampling Methods

Test borings will be performed in accordance with the following: 1) American Society for Testing and
Materials (ASTM) D1586 Standard Test Method for Standard Penetration Test (SPT) and Split Barrel
Sampling of Soils, 2) ASTM D2113 Standard Practice for Rock Core Drilling and Sampling of Rock for
Site Exploration, and 3) and details specified herein. Borings will range from 4 to 8 inches in diameter
and can be drilled by means of a combination of one or more sizes of auger or casing (rotary drilling) and,
if rock is encountered, rock coring using NQ-sized (less than 2-inches in diameter) or HQ-sized (about
2.5 inches in diameter) coring equipment. In general, test borings will extend to a depth of approximately
10 feet below the anticipated bottom of a foundation or until refusal/bedrock is encountered.
If bedrock/refusal is encountered, the boring will be extended a minimum of 10 feet beyond the top of
bedrock/refusal. It is anticipated that drilling depths will range from 25 to 50 feet below ground surface.
Typically, borings of this depth require one-third to one-half of a work day.

Test borings will be extended to the minimum depths specified in boring phase documentation provided
by the GEOR prior to mobilization; the minimum depths provided in GEOR documentation will be in
accordance with requirements stipulated above. Test borings shall not be terminated in material exhibiting
an SPT N-value of less than 7 blows per foot without prior approval from the GEOR.

A minimum of four SPTs (minimum 18 inches inside clear length) will be attempted within the upper
10 feet of each boring; SPTs below 10 feet will be attempted at intervals no greater than 5 feet. Thin-
walled tube samples (i.e., Shelby tubes) may be obtained within anticipated cohesive soils at locations
and/or intervals specified by the GEOR or Quanta Services field representative either prior to drilling
or at the time of drilling. Thin-walled tubes samples will be obtained in accordance with ASTM D1587
“Standard Practice for Thin-Walled Tub Sampling of Soils for Geotechnical Purposes.” Additional
sampling, testing, and documentation requirements during drilling are listed below.

• Drilling in environmentally contaminated areas/conditions is not anticipated. As such, all onsite
personnel will, at a minimum, utilize Level D Personal Protective Equipment (PPE). Level D
